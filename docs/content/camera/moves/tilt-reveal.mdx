```tsx
import { useMemo } from 'react'
import { useVideoConfig } from 'remotion'
import { SceneLayer, SceneStage, secondsToFrameRange } from 'framoo/core'
import { tiltReveal } from 'framoo/camera/moves'

export const TiltRevealPreview = () => {
  const { fps } = useVideoConfig()
  const camera = useMemo(
    () => ({
      moves: [
        tiltReveal({
          ...secondsToFrameRange(0, 3.4, fps),
          focus: { x: 0.64, y: 0.4 },
          zoom: 1.32,
          tiltDegrees: -9,
        }),
      ],
    }),
    [fps],
  )

  return (
    <SceneStage camera={camera}>
      <SceneLayer followsCamera>
        {null /* 背景画像など、カメラと一緒に動かすレイヤーを置く */}
      </SceneLayer>
    </SceneStage>
  )
}
```

## API Reference

| Prop               | Type          | Default                    | Description                              |
| ------------------ | ------------- | -------------------------- | ---------------------------------------- |
| `from`             | `number`      | —                          | 開始フレーム。                           |
| `durationInFrames` | `number`      | —                          | 動きの長さをフレーム数で指定。           |
| `focus`            | `CameraPoint` | —                          | カメラを水平に戻したときの焦点。         |
| `zoom`             | `number`      | 未指定（現在の倍率を維持） | 任意の最終ズーム倍率。                   |
| `tiltDegrees`      | `number`      | `-6`                       | 開始時に加える傾き。角度を度単位で指定。 |
