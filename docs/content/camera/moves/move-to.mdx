```tsx
import { useMemo } from 'react'
import { useVideoConfig } from 'remotion'
import { SceneLayer, SceneStage, secondsToFrameRange } from 'framoo/core'
import { moveTo } from 'framoo/camera/moves'

export const MoveToPreview = () => {
  const { fps } = useVideoConfig()
  const camera = useMemo(
    () => ({
      moves: [
        moveTo({
          ...secondsToFrameRange(0, 4.4, fps),
          to: { x: 0.68, y: 0.38, zoom: 1.28, rotation: -2 },
        }),
      ],
    }),
    [fps],
  )

  return (
    <SceneStage camera={camera}>
      <SceneLayer followsCamera>
        {null /* 背景画像など、カメラと一緒に動かすレイヤーを置く */}
      </SceneLayer>
    </SceneStage>
  )
}
```

## API Reference

| Prop               | Type                  | Default  | Description                                          |
| ------------------ | --------------------- | -------- | ---------------------------------------------------- |
| `from`             | `number`              | —        | 開始フレーム。                                       |
| `durationInFrames` | `number`              | —        | 動きの長さをフレーム数で指定。                       |
| `to`               | `Partial<CameraPose>` | —        | 移動先のカメラ値。省略した値は現在の値を保つ。       |
| `easing`           | `CameraMoveEasing`    | `smooth` | 移動区間の補間。`linear`、`smooth`、`rush`から選択。 |
