```tsx
import { useMemo } from 'react'
import { useVideoConfig } from 'remotion'
import { SceneLayer, SceneStage, secondsToFrameRange } from 'framoo/core'
import { dutchRoll } from 'framoo/camera/moves'

export const DutchRollPreview = () => {
  const { fps } = useVideoConfig()
  const camera = useMemo(
    () => ({
      moves: [
        dutchRoll({
          ...secondsToFrameRange(0, 4.2, fps),
          peakDegrees: 6,
        }),
      ],
    }),
    [fps],
  )

  return (
    <SceneStage camera={camera}>
      <SceneLayer followsCamera>
        {null /* 背景画像など、カメラと一緒に動かすレイヤーを置く */}
      </SceneLayer>
    </SceneStage>
  )
}
```

## API Reference

| Prop               | Type     | Default | Description                                |
| ------------------ | -------- | ------- | ------------------------------------------ |
| `from`             | `number` | —       | 開始フレーム。                             |
| `durationInFrames` | `number` | —       | 動きの長さをフレーム数で指定。             |
| `peakDegrees`      | `number` | `5`     | 左右へ傾ける最大角度。角度を度単位で指定。 |
