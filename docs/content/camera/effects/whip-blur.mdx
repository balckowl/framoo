```tsx
import { useMemo } from 'react'
import { useVideoConfig } from 'remotion'
import { SceneLayer, SceneStage, secondsToFrameRange } from 'framoo/core'
import { whipBlur } from 'framoo/camera/effects'

export const WhipBlurPreview = () => {
  const { fps } = useVideoConfig()
  const camera = useMemo(() => {
    const blurRange = secondsToFrameRange(1.3, 2.1, fps)

    return {
      effects: [
        whipBlur({
          atFrame: blurRange.from,
          durationInFrames: blurRange.durationInFrames,
          amount: 18,
        }),
      ],
    }
  }, [fps])

  return (
    <SceneStage camera={camera}>
      <SceneLayer followsCamera>
        {null /* 背景画像など、カメラと一緒に動かすレイヤーを置く */}
      </SceneLayer>
    </SceneStage>
  )
}
```

## API Reference

| Prop               | Type     | Default            | Description                    |
| ------------------ | -------- | ------------------ | ------------------------------ |
| `atFrame`          | `number` | —                  | ブラーが始まるフレーム。       |
| `durationInFrames` | `number` | 省略時は約0.24秒分 | ブラーが続く長さ。             |
| `amount`           | `number` | `7`                | 最大ブラー半径。ピクセル単位。 |
